// Package tensor implements a minimal N-dimensional float32 tensor:
// shape + strides over a flat buffer, zero-copy views (Reshape, Transpose),
// broadcasting element-wise ops and matrix multiplication.
//
// This is the numeric foundation for the whole course (lecture L1).
package tensor

import (
	"fmt"
	"math"
	"math/rand/v2"
	"strings"
)

// Tensor is a view over a flat float32 buffer.
// Element (i0, i1, ..., in) lives at data[offset + i0*strides[0] + ... + in*strides[n]].
type Tensor struct {
	data    []float32
	shape   []int
	strides []int
	offset  int
}

// New returns a zero-filled contiguous tensor of the given shape.
func New(shape ...int) *Tensor {
	return FromSlice(make([]float32, numel(shape)), shape...)
}

// FromSlice wraps data (without copying) as a contiguous tensor of the given shape.
func FromSlice(data []float32, shape ...int) *Tensor {
	if len(data) != numel(shape) {
		panic(fmt.Sprintf("tensor: FromSlice: %d values do not fit shape %v", len(data), shape))
	}
	return &Tensor{data: data, shape: clone(shape), strides: rowMajorStrides(shape)}
}

// Randn returns a tensor with elements drawn from N(0, 1).
func Randn(rng *rand.Rand, shape ...int) *Tensor {
	t := New(shape...)
	for i := range t.data {
		t.data[i] = float32(rng.NormFloat64())
	}
	return t
}

func (t *Tensor) Shape() []int   { return clone(t.shape) }
func (t *Tensor) Strides() []int { return clone(t.strides) }
func (t *Tensor) Dims() int      { return len(t.shape) }
func (t *Tensor) Size() int      { return numel(t.shape) }

// At returns the element at the given multi-index.
func (t *Tensor) At(idx ...int) float32 { return t.data[t.index(idx)] }

// Set writes v at the given multi-index. Writes through views are visible in the parent.
func (t *Tensor) Set(v float32, idx ...int) { t.data[t.index(idx)] = v }

func (t *Tensor) index(idx []int) int {
	if len(idx) != len(t.shape) {
		panic(fmt.Sprintf("tensor: index %v for shape %v", idx, t.shape))
	}
	off := t.offset
	for d, i := range idx {
		if i < 0 || i >= t.shape[d] {
			panic(fmt.Sprintf("tensor: index %v out of range for shape %v", idx, t.shape))
		}
		off += i * t.strides[d]
	}
	return off
}

// IsContiguous reports whether elements are laid out densely in row-major order.
func (t *Tensor) IsContiguous() bool {
	expected := 1
	for d := len(t.shape) - 1; d >= 0; d-- {
		if t.shape[d] != 1 && t.strides[d] != expected {
			return false
		}
		expected *= t.shape[d]
	}
	return true
}

// Contiguous returns t itself if it is already contiguous, otherwise a dense copy.
func (t *Tensor) Contiguous() *Tensor {
	if t.IsContiguous() {
		return t
	}
	out := New(t.shape...)
	i := 0
	t.each(func(off int) {
		out.data[i] = t.data[off]
		i++
	})
	return out
}

// Data returns the elements in row-major order. For a contiguous tensor the
// returned slice aliases the underlying buffer.
func (t *Tensor) Data() []float32 {
	c := t.Contiguous()
	return c.data[c.offset : c.offset+c.Size()]
}

// Reshape returns a view with a new shape. One dimension may be -1 and is inferred.
// Non-contiguous tensors are copied first (as torch.reshape does).
func (t *Tensor) Reshape(shape ...int) *Tensor {
	shape = clone(shape)
	infer, known := -1, 1
	for d, s := range shape {
		if s == -1 {
			if infer >= 0 {
				panic("tensor: Reshape: more than one -1")
			}
			infer = d
			continue
		}
		known *= s
	}
	if infer >= 0 {
		shape[infer] = t.Size() / known
	}
	if numel(shape) != t.Size() {
		panic(fmt.Sprintf("tensor: Reshape %v -> %v changes size", t.shape, shape))
	}
	c := t.Contiguous()
	return &Tensor{data: c.data, shape: shape, strides: rowMajorStrides(shape), offset: c.offset}
}

// Transpose swaps dimensions a and b. No data is copied: only shape and strides are swapped.
func (t *Tensor) Transpose(a, b int) *Tensor {
	v := &Tensor{data: t.data, shape: clone(t.shape), strides: clone(t.strides), offset: t.offset}
	v.shape[a], v.shape[b] = v.shape[b], v.shape[a]
	v.strides[a], v.strides[b] = v.strides[b], v.strides[a]
	return v
}

// T transposes the last two dimensions (matrix transpose, batched if Dims() > 2).
func (t *Tensor) T() *Tensor {
	n := len(t.shape)
	return t.Transpose(n-2, n-1)
}

// each calls f with the flat offset of every element, in row-major order of t's shape.
func (t *Tensor) each(f func(off int)) {
	eachIndex(t.shape, func(idx []int) {
		off := t.offset
		for d, i := range idx {
			off += i * t.strides[d]
		}
		f(off)
	})
}

// eachIndex iterates over all multi-indices of shape in row-major order
// (like an odometer: the last digit spins fastest).
func eachIndex(shape []int, f func(idx []int)) {
	if numel(shape) == 0 {
		return
	}
	idx := make([]int, len(shape))
	for {
		f(idx)
		d := len(shape) - 1
		for ; d >= 0; d-- {
			idx[d]++
			if idx[d] < shape[d] {
				break
			}
			idx[d] = 0
		}
		if d < 0 {
			return
		}
	}
}

// AllClose reports whether a and b have equal shapes and |a-b| <= atol + rtol*|b| element-wise.
func AllClose(a, b *Tensor, rtol, atol float64) bool {
	if !equalInts(a.shape, b.shape) {
		return false
	}
	ad, bd := a.Data(), b.Data()
	for i := range ad {
		x, y := float64(ad[i]), float64(bd[i])
		if math.Abs(x-y) > atol+rtol*math.Abs(y) {
			return false
		}
	}
	return true
}

func (t *Tensor) String() string {
	var sb strings.Builder
	fmt.Fprintf(&sb, "Tensor%v ", t.shape)
	if t.Size() > 64 {
		sb.WriteString("[...]")
		return sb.String()
	}
	fmt.Fprint(&sb, t.Data())
	return sb.String()
}

func rowMajorStrides(shape []int) []int {
	s := make([]int, len(shape))
	acc := 1
	for d := len(shape) - 1; d >= 0; d-- {
		s[d] = acc
		acc *= shape[d]
	}
	return s
}

func numel(shape []int) int {
	n := 1
	for _, s := range shape {
		n *= s
	}
	return n
}

func clone(s []int) []int { return append([]int(nil), s...) }

func equalInts(a, b []int) bool {
	if len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}
package tensor

import "fmt"

// BroadcastShapes returns the shape produced by broadcasting a and b (NumPy rules):
// shapes are aligned from the right; two dimensions are compatible when equal or one of them is 1.
func BroadcastShapes(a, b []int) ([]int, error) {
	n := max(len(a), len(b))
	out := make([]int, n)
	for i := 1; i <= n; i++ {
		da, db := dimFromRight(a, i), dimFromRight(b, i)
		switch {
		case da == db, db == 1:
			out[n-i] = da
		case da == 1:
			out[n-i] = db
		default:
			return nil, fmt.Errorf("tensor: cannot broadcast %v with %v", a, b)
		}
	}
	return out, nil
}

func dimFromRight(shape []int, i int) int {
	if i > len(shape) {
		return 1
	}
	return shape[len(shape)-i]
}

// broadcastStrides returns strides that let t be read as if it had shape out:
// a broadcast dimension gets stride 0, so the same element is reused.
func (t *Tensor) broadcastStrides(out []int) []int {
	s := make([]int, len(out))
	shift := len(out) - len(t.shape)
	for d := range t.shape {
		if t.shape[d] != 1 {
			s[d+shift] = t.strides[d]
		}
	}
	return s
}

// zipWith applies f element-wise to a and b with broadcasting.
func zipWith(a, b *Tensor, f func(x, y float32) float32) *Tensor {
	shape, err := BroadcastShapes(a.shape, b.shape)
	if err != nil {
		panic(err)
	}
	sa, sb := a.broadcastStrides(shape), b.broadcastStrides(shape)
	out := New(shape...)
	i := 0
	eachIndex(shape, func(idx []int) {
		oa, ob := a.offset, b.offset
		for d, k := range idx {
			oa += k * sa[d]
			ob += k * sb[d]
		}
		out.data[i] = f(a.data[oa], b.data[ob])
		i++
	})
	return out
}

func Add(a, b *Tensor) *Tensor { return zipWith(a, b, func(x, y float32) float32 { return x + y }) }
func Sub(a, b *Tensor) *Tensor { return zipWith(a, b, func(x, y float32) float32 { return x - y }) }
func Mul(a, b *Tensor) *Tensor { return zipWith(a, b, func(x, y float32) float32 { return x * y }) }

// Map returns a new tensor with f applied to every element.
func (t *Tensor) Map(f func(float32) float32) *Tensor {
	out := New(t.shape...)
	i := 0
	t.each(func(off int) {
		out.data[i] = f(t.data[off])
		i++
	})
	return out
}

func (t *Tensor) Scale(s float32) *Tensor { return t.Map(func(x float32) float32 { return x * s }) }

// Sum returns the sum of all elements (accumulated in float64 for accuracy).
func (t *Tensor) Sum() float64 {
	var s float64
	t.each(func(off int) { s += float64(t.data[off]) })
	return s
}

// Dot returns the scalar product of two 1-D tensors of equal length.
func Dot(a, b *Tensor) float32 {
	if a.Dims() != 1 || b.Dims() != 1 || a.shape[0] != b.shape[0] {
		panic(fmt.Sprintf("tensor: Dot needs two equal 1-D tensors, got %v and %v", a.shape, b.shape))
	}
	var s float32
	for i := range a.shape[0] {
		s += a.data[a.offset+i*a.strides[0]] * b.data[b.offset+i*b.strides[0]]
	}
	return s
}

// MatMul multiplies (..., n, k) by (..., k, p) and returns (..., n, p).
// Leading (batch) dimensions are broadcast, so (B, h, T, d) x (B, h, d, T) works,
// and so does (B, T, C) x (C, D) — the same weight matrix applied to every batch element.
func MatMul(a, b *Tensor) *Tensor {
	if a.Dims() < 2 || b.Dims() < 2 {
		panic(fmt.Sprintf("tensor: MatMul needs >=2-D tensors, got %v and %v", a.shape, b.shape))
	}
	n, k := a.shape[a.Dims()-2], a.shape[a.Dims()-1]
	k2, p := b.shape[b.Dims()-2], b.shape[b.Dims()-1]
	if k != k2 {
		panic(fmt.Sprintf("tensor: MatMul inner dims differ: %v x %v", a.shape, b.shape))
	}
	batch, err := BroadcastShapes(a.shape[:a.Dims()-2], b.shape[:b.Dims()-2])
	if err != nil {
		panic(err)
	}

	a, b = a.Contiguous(), b.Contiguous()
	out := New(append(clone(batch), n, p)...)
	sa := (&Tensor{shape: a.shape[:a.Dims()-2], strides: rowMajorStrides(a.shape[:a.Dims()-2])}).broadcastStrides(batch)
	sb := (&Tensor{shape: b.shape[:b.Dims()-2], strides: rowMajorStrides(b.shape[:b.Dims()-2])}).broadcastStrides(batch)

	bi := 0
	eachIndex(batch, func(idx []int) {
		var ia, ib int
		for d, i := range idx {
			ia += i * sa[d]
			ib += i * sb[d]
		}
		am := a.data[a.offset+ia*n*k : a.offset+(ia+1)*n*k]
		bm := b.data[b.offset+ib*k*p : b.offset+(ib+1)*k*p]
		om := out.data[bi*n*p : (bi+1)*n*p]
		MatMulParallel(am, bm, om, n, k, p)
		bi++
	}) // for 2-D inputs batch is empty and this runs exactly once
	return out
}

// The kernels below work on raw row-major slices: a is n×k, b is k×p, out is n×p.
// out must be zeroed by the caller. They exist side by side for the L1 benchmark.

// MatMulNaive is the textbook i-j-k loop. The inner loop walks b down a column
// (stride p), which jumps across memory and misses the CPU cache.
func MatMulNaive(a, b, out []float32, n, k, p int) {
	for i := range n {
		for j := range p {
			var s float32
			for l := range k {
				s += a[i*k+l] * b[l*p+j]
			}
			out[i*p+j] = s
		}
	}
}

// MatMulIKJ reorders the loops so the inner loop walks rows of b and out sequentially:
// out[i,:] += a[i,l] * b[l,:]. Same arithmetic, cache-friendly access.
func MatMulIKJ(a, b, out []float32, n, k, p int) {
	for i := range n {
		orow := out[i*p : (i+1)*p]
		for l := range k {
			ail := a[i*k+l]
			brow := b[l*p : (l+1)*p]
			for j, blj := range brow {
				orow[j] += ail * blj
			}
		}
	}
}

const tile = 64

// MatMulTiled processes the matrices in tile×tile blocks so that the working set
// of the three inner loops fits in L1/L2 cache.
func MatMulTiled(a, b, out []float32, n, k, p int) {
	matMulTiledRows(a, b, out, n, k, p, 0, n)
}

func matMulTiledRows(a, b, out []float32, n, k, p, rowFrom, rowTo int) {
	for i0 := rowFrom; i0 < rowTo; i0 += tile {
		iMax := min(i0+tile, rowTo)
		for l0 := 0; l0 < k; l0 += tile {
			lMax := min(l0+tile, k)
			for j0 := 0; j0 < p; j0 += tile {
				jMax := min(j0+tile, p)
				for i := i0; i < iMax; i++ {
					orow := out[i*p+j0 : i*p+jMax]
					for l := l0; l < lMax; l++ {
						ail := a[i*k+l]
						brow := b[l*p+j0 : l*p+jMax]
						for j, blj := range brow {
							orow[j] += ail * blj
						}
					}
				}
			}
		}
	}
}

// MatMulParallel splits the rows of out between GOMAXPROCS goroutines, each running
// the tiled kernel. Rows are independent, so no synchronization on out is needed.
func MatMulParallel(a, b, out []float32, n, k, p int) {
	if n*k*p < 64*64*64 {
		MatMulIKJ(a, b, out, n, k, p)
		return
	}
	workers := min(runtime.GOMAXPROCS(0), (n+tile-1)/tile)
	chunk := ((n+workers-1)/workers + tile - 1) / tile * tile
	var wg sync.WaitGroup
	for from := 0; from < n; from += chunk {
		to := min(from+chunk, n)
		wg.Go(func() { matMulTiledRows(a, b, out, n, k, p, from, to) })
	}
	wg.Wait()
}
